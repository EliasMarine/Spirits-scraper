## Phase 4: Brand Normalization System (Priority: MEDIUM)

### 4.1 Create Brand Service
**File**: `src/services/brand-service.ts`

```typescript
export class BrandService {
  private brandMappings = new Map<string, string>();
  
  constructor() {
    this.loadBrandMappings();
  }
  
  private loadBrandMappings() {
    // Load from database or config
    this.brandMappings.set('angels envy', 'Angel\'s Envy');
    this.brandMappings.set('four roses', 'Four Roses');
    // ... etc
  }
  
  normalizeBrand(brand: string): string {
    const normalized = brand.toLowerCase().trim();
    return this.brandMappings.get(normalized) || this.titleCase(brand);
  }
}
```

### 4.2 Database Brand Table
Create a `brand_mappings` table:
```sql
CREATE TABLE brand_mappings (

  normalized_brand TEXT NOT NULL,
  brand_id UUID REFERENCES brands(id),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

```

## Phase 5: Description Quality Enhancement (Priority: MEDIUM)

### 5.1 Description Processor
**File**: `src/services/description-processor.ts`

```typescript
export class DescriptionProcessor {
  cleanDescription(desc: string): string | null {
    // Remove search snippets
    desc = desc.replace(/Find the best local for.*/i, '');
    desc = desc.replace(/Avg \(ex-tax\).*/i, '');
    
    // Remove navigation elements
    desc = desc.replace(/Home\s*>\s*[\w\s>]+/g, '');
    
    // Remove related products sections
    desc = desc.replace(/Related Products:.*/is, '');
    desc = desc.replace(/Similar Products:.*/is, '');
    
    // Minimum quality check
    if (desc.length < 50) return null;
    
    return desc.trim();
  }
}
```
