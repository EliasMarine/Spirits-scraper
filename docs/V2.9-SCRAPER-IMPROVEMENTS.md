```typescript
// services/spirit-extractor.ts
private isInvalidProductUrl(url: string): boolean {
  const invalidPaths = [
    '/recipes/',
    '/cocktails/',
    '/collections/',
    '/blog/',
    '/articles/',
    '/menu/',
    '/events/',
    '/podcast',
    '/category/',
    '/tag/',
  ];
  
  const invalidDomains = [
    'podcasts.apple.com',
    'spotify.com',
    'youtube.com',
    'instagram.com',
    'facebook.com',
  ];
  
  return invalidPaths.some(path => url.includes(path)) ||
         invalidDomains.some(domain => url.includes(domain));
}
```

```typescript
// services/content-parser.ts
private extractPrice(text: string): string | null {
  // Skip if contains disqualifying patterns
  const skipPatterns = [
    /\b(19\d{2}|20\d{2})\b/, // Years
    /\b\d+\s*(year|yr|age)\b/i, // Ages
    /\b\d+\s*proof\b/i, // Proof values
    /\bbatch\s*#?\s*\d+/i, // Batch numbers
  ];
  
  if (skipPatterns.some(pattern => pattern.test(text))) {
    return null;
  }
  
  // Enhanced price patterns
  const pricePatterns = [
    // Currency symbols
    /(?:USD|EUR|GBP|£|€|\$)\s*(\d{1,4}(?:\.\d{2})?)/,
    /(\d{1,4}(?:\.\d{2})?)\s*(?:USD|EUR|GBP|dollars?)/i,
    
    // Price ranges
    /\$(\d{1,4}(?:\.\d{2})?)\s*[-–]\s*\$?\d{1,4}/,
    /between\s+\$(\d{1,4}(?:\.\d{2})?)\s+and/i,
    
    // Context-specific
    /price:?\s*\$?(\d{1,4}(?:\.\d{2})?)/i,
    /costs?\s*\$?(\d{1,4}(?:\.\d{2})?)/i,
    /retail:?\s*\$?(\d{1,4}(?:\.\d{2})?)/i,
    /msrp:?\s*\$?(\d{1,4}(?:\.\d{2})?)/i,
    
    // E-commerce patterns
    /add\s+to\s+cart.*?\$(\d{1,4}(?:\.\d{2})?)/i,
    /buy\s+now.*?\$(\d{1,4}(?:\.\d{2})?)/i,
  ];
  
  for (const pattern of pricePatterns) {
    const match = text.match(pattern);
    if (match) {
      const price = parseFloat(match[1]);
      if (price >= 5 && price <= 5000) {
        return `$${price.toFixed(2)}`;
      }
    }
  }
  
  return null;
}
```

```typescript
// Extract from JSON-LD
private extractFromJsonLd($: CheerioStatic): ParsedContent {
  const data: ParsedContent = {};
  
  $('script[type="application/ld+json"]').each((_, elem) => {
    try {
      const jsonLd = JSON.parse($(elem).text());
      
      // Product schema
      if (jsonLd['@type'] === 'Product') {
        data.name = jsonLd.name;
        data.brand = jsonLd.brand?.name || jsonLd.manufacturer?.name;
        data.description = jsonLd.description;
        
        // Price extraction
        if (jsonLd.offers) {
          const offers = Array.isArray(jsonLd.offers) ? jsonLd.offers[0] : jsonLd.offers;
          data.price = offers.price || offers.lowPrice || offers.highPrice;
          data.availability = offers.availability;
        }
        
        // Additional properties
        data.sku = jsonLd.sku;
        data.gtin = jsonLd.gtin13 || jsonLd.gtin12;
        data.mpn = jsonLd.mpn;
      }
      
      // Breadcrumb for category
      if (jsonLd['@type'] === 'BreadcrumbList') {
        const items = jsonLd.itemListElement || [];
        const categories = items.map(item => item.name).filter(Boolean);
        data.category = categories[categories.length - 1];
      }
    } catch (e) {
      // Invalid JSON, skip
    }
  });
  
  return data;
}
```

## 🎯 Priority 3: Smarter Quality Scoring

### Adjusted Quality Weights
```typescript
// services/pre-storage-validator.ts
private calculateQualityScore(spiritData: any): number {
  let score = 100;
  
  // Price is a strong quality signal
  if (!spiritData.price) {
    score -= 20; // Increased from 15
  } else {
    score += 10; // Bonus for having price
  }
  
  // Category classification
  if (spiritData.type === 'Other' || !spiritData.type) {
    score -= 25;
  } else if (spiritData.type !== 'Spirit') {
    score += 10; // Bonus for specific category
  }
  
  // Brand quality
  if (!spiritData.brand || spiritData.brand === 'Unknown') {
    score -= 15;
  } else if (spiritData.brand.length > 3) {
    score += 5;
  }
  
  // Name quality
  if (spiritData.name.length < 15) {
    score -= 20;
  }
  
  // Description quality
  if (!spiritData.description) {
    score -= 10;
  } else if (spiritData.description.length > 50) {
    score += 5;
  }
  
  // Additional metadata bonuses
  if (spiritData.abv) score += 5;
  if (spiritData.proof) score += 5;
  if (spiritData.age) score += 10;
  if (spiritData.distillery) score += 10;
  if (spiritData.region) score += 5;
  
  return Math.max(0, Math.min(100, score));
}
```

```typescript
private getMinimumQualityScore(spiritData: any): number {
  // Base threshold
  let threshold = 55;
  
  // Lower threshold if has price (strong signal)
  if (spiritData.price && spiritData.price > 0) {
    threshold = 45;
  }
  
  // Lower threshold for special categories
  const specialCategories = ['Cognac', 'Japanese Whisky', 'Mezcal', 'Armagnac'];
  if (specialCategories.includes(spiritData.type)) {
    threshold -= 5;
  }
  
  // Higher threshold for generic entries
  if (spiritData.type === 'Other' || !spiritData.brand) {
    threshold += 10;
  }
  
  return threshold;
}
```

```typescript
// services/batch-processor.ts
private async processBatchItem(item: any): Promise<void> {
  // Early URL validation
  if (this.isInvalidProductUrl(item.link)) {
    logger.warn(`Skipping invalid URL pattern: ${item.link}`);
    this.stats.skipped++;
    return;
  }
  
  // Early title validation
  if (this.containsNonProductPatterns(item.title)) {
    logger.warn(`Skipping non-product title: ${item.title}`);
    this.stats.skipped++;
    return;
  }
  
  // Process normally...
  const result = await this.processSpirit(item);
  
  // Post-processing validation
  if (result.qualityScore < 30) {
    logger.warn(`Skipping very low quality result: ${result.name} (score: ${result.qualityScore})`);
    this.stats.rejected++;
    return;
  }
  
  // Store only quality results
  await this.storeSpirit(result);
}
```
