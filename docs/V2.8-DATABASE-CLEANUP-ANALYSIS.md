### 2. **Enhanced Price Extraction**
```typescript
// Add more sophisticated price extraction
const extractPriceFromStructuredData = (jsonLd: any): number | null => {
  if (jsonLd.offers?.price) return parseFloat(jsonLd.offers.price);
  if (jsonLd.offers?.lowPrice) return parseFloat(jsonLd.offers.lowPrice);
  if (jsonLd.priceRange) {
    const match = jsonLd.priceRange.match(/\$?(\d+(?:\.\d{2})?)/);
    if (match) return parseFloat(match[1]);
  }
  return null;
};
```

## SQL Monitoring Queries

### Check Data Quality Trends
```sql
-- Daily quality metrics
SELECT 
    DATE(created_at) as date,
    COUNT(*) as total_spirits,
    COUNT(CASE WHEN price IS NOT NULL THEN 1 END) as with_price,
    ROUND(AVG(data_quality_score), 1) as avg_quality,
    COUNT(CASE WHEN data_quality_score < 30 THEN 1 END) as very_low_quality
FROM spirits
WHERE created_at >= NOW() - INTERVAL '30 days'
GROUP BY DATE(created_at)
ORDER BY date DESC;
```

### Identify Problem Patterns
```sql
-- Find common patterns in low-quality entries
SELECT 
    SUBSTRING(name FROM 1 FOR 50) as name_sample,
    COUNT(*) as occurrences,
    AVG(data_quality_score) as avg_score
FROM spirits
WHERE data_quality_score < 40
    AND created_at >= NOW() - INTERVAL '7 days'
GROUP BY SUBSTRING(name FROM 1 FOR 50)
HAVING COUNT(*) > 1
ORDER BY occurrences DESC;
```
