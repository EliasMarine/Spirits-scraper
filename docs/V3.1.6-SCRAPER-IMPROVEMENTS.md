```typescript
interface ValidationRule {
  name: string;
  check: (spirit: SpiritData) => boolean;
  errorMessage: string;
  severity: 'ERROR' | 'WARNING';
}

const CRITICAL_VALIDATION_RULES: ValidationRule[] = [
  {
    name: 'URL_REQUIRED',
    check: (s) => !!(s.url && s.url.trim()),
    errorMessage: 'URL is required for all spirits',
    severity: 'ERROR'
  },
  {
    name: 'VALID_PRODUCT_NAME',
    check: (s) => isValidProductName(s.name),
    errorMessage: 'Product name appears to be article title or search query',
    severity: 'ERROR'
  },
  {
    name: 'CATEGORY_NOT_OTHER',
    check: (s) => s.category !== 'Other',
    errorMessage: 'Category detection failed - defaulted to Other',
    severity: 'WARNING'
  },
  {
    name: 'DESCRIPTION_QUALITY',
    check: (s) => s.description && s.description.length >= 20,
    errorMessage: 'Description missing or too short',
    severity: 'WARNING'
  }
];

function validateSpirit(spirit: SpiritData): ValidationResult {
  const errors: string[] = [];
  const warnings: string[] = [];
  
  for (const rule of CRITICAL_VALIDATION_RULES) {
    if (!rule.check(spirit)) {
      if (rule.severity === 'ERROR') {
        errors.push(rule.errorMessage);
      } else {
        warnings.push(rule.errorMessage);
      }
    }
  }
  
  return {
    isValid: errors.length === 0,
    errors,
    warnings,
    canStore: errors.length === 0 // Block storage for any errors
  };
}
```

```typescript
// MONITOR: Real-time scraping quality
class ScrapingQualityMonitor {
  private qualityThresholds = {

      averageQualityScore: spirits.reduce((sum, s) => sum + (s.data_quality_score || 0), 0) / spirits.length,

      logger.error('QUALITY ALERT: Scraping quality below thresholds', { report, issues });
```
