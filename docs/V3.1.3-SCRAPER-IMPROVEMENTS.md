### 4. Enhanced Source Quality Scoring

**Problem**: Low-quality sources producing bad data

**Solution**: Source reputation system
```javascript
class SourceQualityScorer {
  scoreSource(domain: string, html: string): number {
    let score = 100;
    
    // Positive signals
    if (this.hasStructuredData(html)) score += 20;
    if (this.hasPriceInMultiplePlaces(html)) score += 15;
    if (this.hasProductSchema(html)) score += 25;
    if (this.trustedDomains.has(domain)) score += 30;
    
    // Negative signals
    if (this.hasBlogPatterns(html)) score -= 40;
    if (this.hasReviewPatterns(html)) score -= 30;
    if (this.hasNavigationText(html)) score -= 20;
    
    return Math.max(0, Math.min(100, score));
  }
}
```

### 5. Name Cleaning Pipeline

**Problem**: Volume suffixes, spacing issues

**Solution**: Comprehensive cleaning pipeline
```javascript
class NameCleaner {
  clean(name: string): string {
    return name
      // Remove volume suffixes
      .replace(/\s+(Bottle|70\s*cl|750\s*ml|1\s*L|1L)$/i, '')
      // Fix spacing
      .replace(/\bL\s+L\b/g, 'll')
      .replace(/\bLl\b/g, 'll')
      // Fix possessives
      .replace(/['']S$/i, "'s")
      // Remove store names
      .replace(/\s+at\s+.*(wine|liquor|spirits).*/i, '')
      // Normalize whitespace
      .replace(/\s+/g, ' ')
      .trim();
  }
}
```

### 8. Intelligent Retry Logic

**Problem**: Single-pass extraction misses data

**Solution**: Targeted retry for missing fields
```javascript
class SmartExtractor {
  async extract(html: string, url: string): Promise<SpiritData> {
    let data = await this.initialExtraction(html);
    
    // Retry for critical missing fields
    if (!data.price) {
      data.price = await this.deepPriceExtraction(html, url);
    }
    
    if (!data.brand || this.isInvalidBrand(data.brand)) {
      data.brand = await this.enhancedBrandExtraction(html, data.name);
    }
    
    return data;
  }
}
```
