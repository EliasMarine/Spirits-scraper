async function checkDuplicate(spirit: SpiritData): Promise<boolean> {

### 3. BRAND NORMALIZATION

#### Improved Normalization:

const BRAND_MAPPINGS = {
  "angel's envy": "Angel's Envy",
  "angels envy": "Angel's Envy",
  "four roses": "Four Roses",
  "widow jane": "Widow Jane",
  "maker's mark": "Maker's Mark",
  "buffalo trace": "Buffalo Trace",
  "jack daniel's": "Jack Daniel's",
  // Add more mappings
};

function normalizeBrand(brand: string): string {
  if (!brand) return "Unknown";
  
  const cleaned = brand
    .toLowerCase()
    .trim()
    .replace(/['']/g, "'")
    .replace(/\s+/g, " ");
    
  return BRAND_MAPPINGS[cleaned] || 
         brand.split(' ')
               .map(word => word.charAt(0).toUpperCase() + word.slice(1))
               .join(' ');
}
```

function cleanDescription(desc: string): string | null {
  if (!desc) return null;
  
  // Remove common search result patterns
  const cleaned = desc
    .replace(/find the best local for.*/i, '')
    .replace(/avg \(ex-tax\).*/i, '')
    .replace(/find and shop from.*/i, '')
    .replace(/similar products.*/i, '')
    .replace(/related products.*/i, '')
    .trim();
    
  // Minimum quality threshold
  if (cleaned.length < 50) return null;
  
  // Remove if it's just navigation text
  if (/^(home|shop|products|whiskey|bourbon|scotch)$/i.test(cleaned)) {
    return null;
  }
  
  return cleaned;
}
```

function extractPrice(text: string): number | null {

  const patterns = [
    /\$(\d+(?:\.\d{2})?)/,          // $123.45
    /USD\s*(\d+(?:\.\d{2})?)/i,     // USD 123.45
    /(\d+(?:\.\d{2})?)\s*USD/i,     // 123.45 USD
    /price:?\s*\$?(\d+(?:\.\d{2})?)/i, // Price: $123.45
    /MSRP:?\s*\$?(\d+(?:\.\d{2})?)/i,  // MSRP: $123.45
  ];
  
  for (const pattern of patterns) {
    const match = text.match(pattern);
    if (match) {
      const price = parseFloat(match[1]);
      // Sanity check
      if (price > 10 && price < 50000) {
        return price;
      }
    }
  }
  
  return null;
}

### 7. VALIDATION PIPELINE

#### Comprehensive Validation:
```typescript
interface ValidationResult {
  isValid: boolean;
  issues: string[];
  suggestions: Record<string, any>;
}

function validateSpirit(spirit: SpiritData): ValidationResult {
  const issues: string[] = [];
  const suggestions: Record<string, any> = {};
  
  // Name validation
  if (!spirit.name || spirit.name.length < 5) {
    issues.push("Name too short or missing");
  }
  if (spirit.name && spirit.name.length > 100) {
    issues.push("Name too long, likely contains description");
    suggestions.name = spirit.name.substring(0, 100).trim();
  }
  
  // Brand validation
  if (!spirit.brand || spirit.brand === "Unknown") {
    issues.push("Brand missing or unknown");
  }
  
  // Type validation
  const validTypes = ['Bourbon', 'Scotch', 'Whiskey', 'Rum', 'Tequila', 'Gin', 'Vodka', 'Cognac', 'Brandy'];
  if (!spirit.type || !validTypes.includes(spirit.type)) {
    issues.push("Invalid or missing spirit type");
  }
  
  // ABV validation
  if (spirit.abv && (spirit.abv < 20 || spirit.abv > 70)) {
    issues.push("ABV out of reasonable range");
  }
  
  // Price validation
  if (spirit.price && (spirit.price < 10 || spirit.price > 50000)) {
    issues.push("Price out of reasonable range");
  }
  
  // Check for non-spirit content
  if (isNonSpiritContent(spirit.name, spirit.description)) {
    issues.push("Appears to be non-spirit content");
  }
  
  return {
    isValid: issues.length === 0,
    issues,
    suggestions
  };
}
```
